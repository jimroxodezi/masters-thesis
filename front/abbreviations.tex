\section*{Symbols}
\begin{description}[style=sameline, leftmargin=3.6cm, labelwidth=3.3cm, font=\normalfont, itemsep=0pt]
  \item[$\mathcal{H}_0$, $\mathcal{H}_1$] Null hypothesis (noise only) and alternative hypothesis (signal present)
  \item[$y(n)$]     Received complex baseband sample $n$
  \item[$s(n)$, $w(n)$] Signal and noise components of $y(n)$
  \item[$N$]        Number of samples in one sensing decision
  \item[$T$]        Test statistic (average power for the energy detector)
  \item[$\lambda$]  Decision threshold
  \item[$\Pd$]  Probability of detection
  \item[$\Pfa$] Probability of false alarm
  \item[$\sigma^2$] Noise variance (noise power)
  \item[$\sigma_n^2$] Nominal noise power
  \item[$\sigma_s^2$] Signal variance (signal power)
  \item[$\gamma$]   Signal-to-noise ratio, $\sigma_s^2/\sigma^2$
  \item[$\rho$, $\delta$] Noise-uncertainty factor, $\rho = 10^{\delta/10}$, for an uncertainty of $\delta$ decibels
  \item[$F_{\chi^2_{2N}}$] Cumulative distribution function of a chi-square variable with $2N$ degrees of freedom
  \item[$h(m)$]     STFT window
  \item[$L$, $R$, $K$] STFT window length, frame spacing (hop) and number of frequency bins
  \item[$f_s$]      Sampling rate
  \item[$X(\ell, k)$, $S(\ell, k)$] STFT and spectrogram (dB) at frame $\ell$ and bin $k$
  \item[$f$, $g$]   Encoder and decoder of an autoencoder
  \item[$D$, $d$]   Number of input elements and code dimension
  \item[$e(\mathbf{x})$] Reconstruction-error score of input $\mathbf{x}$
  \item[$\mathbf{W}$] PCA projection matrix (rows: leading eigenvectors)
  \item[$g_1$, $g_2$] Primary and secondary fixed receiver gain settings
  \item[$f_\mathrm{off}$, $W_\mathrm{occ}$] Offset of the C\textsubscript{offset} window from the carrier centre; occupied width of the carrier
  \item[$P_s$, $P_n$, $\hat{P}_{n,C}$] Signal power, noise power, and the session's median idle noise power at the same gain
  \item[$\alpha$]   Scale factor applied to an active frame in the controlled-SNR set
  \item[$\gamma_\mathrm{eff}$] Effective SNR of a controlled-SNR mixture
  \item[$\hat{\sigma}^2$, $M$, $\kappa$] Running noise estimate, its window (frames) and the threshold of the adaptive energy detector
  \item[$d_u(n)$, $p_u(m)$] PSS Zadoff--Chu sequence with root $u$, and its 128-sample time-domain template
  \item[$\Lambda_\mathrm{PSS}$] Score of the \syncsignal{} correlator
  \item[$\Delta\Pd$] Difference in detection probability between two detectors
\end{description}
